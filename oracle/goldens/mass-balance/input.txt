# weight recomputes its shape inertia; an explicit inertia is kept. Moving
# one shifts the CG, which the next run reports as a delta.

run
set inertia/pointmass-weight-lbs[4] 30
set inertia/pointmass-weight-lbs[6] 0
run
set inertia/pointmass-location-X-inches[2] 75.5
set inertia/pointmass-location-Y-inches[3] -4
set inertia/pointmass-location-Z-inches[7] 12
run
run
init
run
set inertia/pointmass-weight-lbs 1
set inertia/weight-lbs 5
set inertia/cg-x-in 5
run
